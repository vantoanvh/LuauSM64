--!strict
--!optimize 2

--[[
	Native symbols of the ROM scripts: the function or C global behind each address stored in script data (CALL_NATIVE of behavior
	scripts, CALL and CALL_LOOP of level scripts, GEO_ASM and the other geo commands that hold a function, SPAWN_WATER_DROPLET
	that holds a WaterDropletParams).
	Generated by native/tools/gen_native_calls.luau (rom-function-pointer-table, script-data-pointer-slots); do not edit.

	Slots
		A slot is a word of the script data, named by the ROM start of the block that holds it (the `from` of the LOAD_RAW that
		loaded the segment: behavior scripts, the level scripts and geo layouts of a level, the actor geo layouts of a group) and
		its byte offset inside that block, which is the offset of its segmented address (Rom.BlockOf gives both from the address
		of the slot in Memory).

	Design Notes
		FunctionAt and DataAt find the symbol by name among the members of the registered modules the first time a name is used;
		its module must have loaded by then.
]]

const Registry = require("../../groups/registry")

const Symbols: { [string]: unknown } = {}
const NativeCalls = {
	-- Block ROM start, then slot offset, to the name of the C function or global
	Blocks = {
		-- BLOCKS
	},
}

const function NameOf(BlockStart: number, Offset: number): string?
	const Slots = NativeCalls.Blocks[BlockStart]

	return if Slots == nil then nil else Slots[Offset]
end

const function SymbolAt(BlockStart: number, Offset: number): unknown
	const Name = NameOf(BlockStart, Offset)

	if Name == nil then
		error(`no native symbol at offset {Offset} of the block at ROM {BlockStart}`)
	end

	local Symbol = Symbols[Name]

	if Symbol == nil then
		for _, Module in Registry.Members :: { [string]: { [string]: unknown } } do
			const Member = Module[Name]

			if Member ~= nil then
				Symbol = Member

				break
			end
		end

		if Symbol == nil then
			error(`native symbol {Name} is not in a registered module`)
		end

		Symbols[Name] = Symbol
	end

	return Symbol
end

-- The name of the C function or global at a slot of script data; nil where the block has no slot there
function NativeCalls.NameAt(BlockStart: number, Offset: number): string?
	return NameOf(BlockStart, Offset)
end

-- The Luau function at a slot of script data
function NativeCalls.FunctionAt(BlockStart: number, Offset: number): (...unknown) -> ...unknown
	return SymbolAt(BlockStart, Offset) :: (...unknown) -> ...unknown
end

-- The Luau value of the C global at a slot of script data
function NativeCalls.DataAt(BlockStart: number, Offset: number): unknown
	return SymbolAt(BlockStart, Offset)
end

return NativeCalls
